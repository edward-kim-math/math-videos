\begin{frame}
\frametitle{Solve for $P$ in $A = P + Prt$}
\pause $A = P(1 + rt)$ \vskip4pt
\pause $\frac{A}{1 + rt}=P$
\end{frame}

\begin{frame}
\frametitle{Solve for $T$ in $T = \frac{3(4-Tr)}{2}$}
\pause $2T = 3(4-Tr)$ \vskip4pt
\pause $2T = 12 - 3Tr$ \vskip4pt
\pause $2T + 3Tr = 12$ \vskip4pt
\pause $T(2 + 3r) = 12$ \vskip4pt
\pause $T = \frac{12}{2 + 3r}$
\end{frame}

\begin{frame}
\frametitle{Completely factor $3x^3-6x^2-4x+8$}
\pause $=3x^2(\onslide<3->{x}-\onslide<4->{2})-4(\onslide<5->{x}\onslide<6->{-2})$ \vskip4pt
\onslide<7->{$=(\onslide<8->{3x^2}-\onslide<9->{4})(x-2)$}
\end{frame}

\begin{frame}

\onslide<8->{$=(\onslide<9->{2x^2}-\onslide<10->{1})(x-2)$}
\end{frame}

\begin{frame}
\frametitle{Completely factor $5x^5-10x^4+15x^3-30x^2$}

\onslide<7->{$=5x^2(\onslide<8->{x^2}(\onslide<9->{x}-\onslide<10->{2})+\onslide<11->{3}(\onslide<12->{x}-\onslide<13->{2}))$} \vskip4pt
\onslide<14->{$=5x^2(\onslide<15->{x^2}+\onslide<16->{3})(x-2)$}
\end{frame}
